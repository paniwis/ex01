    function texto($valor) {
        return iconv("UTF-8", "windows-1252//TRANSLIT", (string)$valor);
    }
    
    function generarPDF($titulo, $encabezados, $anchos, $campos, $res, $extra = "") {
        $pdf = new FPDF();
        $pdf->AddPage();

        $pdf->SetFont("Arial", "B", 14);
        $pdf->Cell(190, 10, texto($titulo), 0, 1, "C");
        $pdf->SetFont("Arial", "", 10);
        $pdf->Cell(190, 8, "Fecha: " . date("d/m/Y"), 0, 1, "C");
        $pdf->Ln(3);

        $pdf->SetFont("Arial", "B", 10);
        for ($i = 0; $i < count($encabezados); $i++) {
            $salto = ($i == count($encabezados) - 1) ? 1 : 0;
            $pdf->Cell($anchos[$i], 8, texto($encabezados[$i]), 1, $salto, "C");
        }

        $pdf->SetFont("Arial", "", 10);
        while ($fila = $res->fetch_assoc()) {
            for ($i = 0; $i < count($campos); $i++) {
                $salto = ($i == count($campos) - 1) ? 1 : 0;
                $pdf->Cell($anchos[$i], 8, texto($fila[$campos[$i]]), 1, $salto, "C");
            }
        }

        $pdf->Ln(3);
        $pdf->Cell(190, 8, "Total de registros: " . $res->num_rows, 0, 1, "L");
        if ($extra != "") {
            $pdf->SetFont("Arial", "B", 10);
            $pdf->Cell(190, 8, texto($extra), 0, 1, "L");
        }
        $pdf->Output("I", "reporte.pdf");
        exit;   
    }

// ---- PRODUCTOS: guardar (nuevo o editar) ----
if (isset($_POST["guardar_producto"])) {
    if ($_POST["original"] == "") {
        $stmt = $conn->prepare("INSERT INTO productos VALUES (?, ?, ?, ?)");
        $stmt->bind_param("ssdi", $_POST["codigo"], $_POST["nombre"], $_POST["precio"], $_POST["stock"]);
    } else {
        $stmt = $conn->prepare("UPDATE productos SET codigo=?, nombre=?, precio=?, stock=? WHERE codigo=?");
        $stmt->bind_param("ssdis", $_POST["codigo"], $_POST["nombre"], $_POST["precio"], $_POST["stock"], $_POST["original"]);
    }
    if (!$stmt->execute()) {
        die("Error al guardar el producto: " . $conn->error . " <a href='index.php'>Volver</a>");
    }
    header("Location: index.php");
    exit;
}

// ---- PEDIDOS: registrar (cliente, producto, cantidad y fecha) ----
if (isset($_POST["registrar_pedido"])) {
    $cedula = $_POST["cedula"];
    $codigo = $_POST["codigo"];
    $cantidad = $_POST["cantidad"];

    // 1. Revisar que haya stock
    $stmt = $conn->prepare("SELECT stock FROM productos WHERE codigo=?");
    $stmt->bind_param("s", $codigo);
    $stmt->execute();
    $producto = $stmt->get_result()->fetch_assoc();
    if ($producto["stock"] < $cantidad) {
        die("No hay stock suficiente. <a href='index.php'>Volver</a>");
    }

    // 2. Guardar el pedido (el id_pedido lo pone MySQL solo)
    $stmt = $conn->prepare("INSERT INTO pedidos (cedula, codigo, cantidad, fecha) VALUES (?, ?, ?, ?)");
    $stmt->bind_param("ssis", $cedula, $codigo, $cantidad, $_POST["fecha"]);
    $stmt->execute();

    // 3. Descontar el stock
    $stmt = $conn->prepare("UPDATE productos SET stock = stock - ? WHERE codigo=?");
    $stmt->bind_param("is", $cantidad, $codigo);
    $stmt->execute();

    header("Location: index.php");
    exit;
}

// ---- PEDIDOS: eliminar (devuelve el stock) ----
if (isset($_GET["eliminar_pedido"])) {
    $id_pedido = $_GET["eliminar_pedido"];

    // 1. Devolver el stock (antes de borrar, para poder leer la cantidad y el producto)
    $stmt = $conn->prepare("UPDATE productos p, pedidos pe SET p.stock = p.stock + pe.cantidad
                            WHERE p.codigo = pe.codigo AND pe.id_pedido=?");
    $stmt->bind_param("i", $id_pedido);
    $stmt->execute();

    // 2. Borrar el pedido
    $stmt = $conn->prepare("DELETE FROM pedidos WHERE id_pedido=?");
    $stmt->bind_param("i", $id_pedido);
    $stmt->execute();

    header("Location: index.php");
    exit;
}
